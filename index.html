// ========================================== // FITUR AUTO-SCAN (TANPA KLIK KOTAK INPUT) // ========================================== let barcodeBuffer = ""; let lastKeyTime = Date.now(); document.addEventListener('keypress', function(e) { // 1. Pastikan fitur ini hanya aktif saat menu "Scan" terbuka if (document.getElementById('view-scan').classList.contains('hidden')) return; // 2. Abaikan jika admin sedang mengklik/mengubah pilihan dropdown "Pilih Sesi" if (e.target.tagName === 'SELECT') return; const currentTime = Date.now(); // 3. Scanner gun mengetik huruf sangat cepat (jeda di bawah 100ms). // Jika ketikan lambat (di atas 100ms), sistem anggap itu manusia, reset buffer. if (currentTime - lastKeyTime > 100) { barcodeBuffer = ""; } lastKeyTime = currentTime; // 4. Alat scanner otomatis menekan "Enter" di akhir tembakan barcode if (e.key === 'Enter') { if (barcodeBuffer !== "") { e.preventDefault(); prosesScanDatabase(barcodeBuffer); // Jalankan absen ke database barcodeBuffer = ""; // Bersihkan memori buffer document.getElementById('input-scanner').value = ""; // Kosongkan layar kotak } } else { // Kumpulkan angka/huruf dari tembakan scanner barcodeBuffer += e.key; // Tampilkan animasinya di kotak input agar terlihat seperti diketik otomatis document.getElementById('input-scanner').value = barcodeBuffer; } });
